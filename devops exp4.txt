Procedure:

1. Create folder in desktop "minikube" and open with vs code
2. create files and paste the code (app, requirements, dockerfile, deployment, services)
3. build image and run container(terminal 1)
	- docker build -t "app name" .
	- docker run -p 5000:5000 "app name" 
4. verify output localhost:5000
5. start minikube (terminal 2)
	- minikube start --driver=docker                                                                                        
  	- kubectl get nodes                                                                                                     
   	- kubectl get pods                                                                                                      
  	- kubectl run nginx-pod --image=nginx                                                                                                                                                                                             	- kubectl get pods                                                                                                     
	- kubectl expose pod nginx-pod --type=NodePort --port=80                                                                
	- minikube service nginx-pod(nginx output)
6. dashboard(terminal 3)
	- minikube dashboard


minikube/app.py

from flask import Flask

app = Flask(__name__)

@app.route('/')
def home():
    return "Hello from Dockerized Flask App!"

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)

minikube/requirements.txt

flask

minikube/Dockerfile

FROM python:3.9-slim

WORKDIR /app

COPY requirements.txt .
RUN pip install -r requirements.txt

COPY app.py .

EXPOSE 5000

CMD ["python", "app.py"]


minikube/deployment.yaml

apiVersion: apps/v1
kind: Deployment
metadata:
  name: flask-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: flask-app
  template:
    metadata:
      labels:
        app: flask-app
    spec:
      containers:
      - name: flask-container
        image: suni484/flask-k8:latest
        ports:
        - containerPort: 5000


minikube/services.yaml

apiVersion: v1
kind: Service
metadata:
  name: flask-service
spec:
  type: NodePort
  selector:
    app: flask-app
  ports:
    - protocol: TCP
      port: 5000
      targetPort: 5000
      nodePort: 30007
